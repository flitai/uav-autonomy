cmake_minimum_required(VERSION 3.31)
project(UavAutonomy LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native MSVC x64 is required")
endif()
if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "19.44.35229")
    message(FATAL_ERROR "Compiler differs from the validated T01 baseline")
endif()
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Validated configuration" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_GENERATOR STREQUAL "Ninja")
    set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_LIBRARIES OFF)
endif()
option(UXAS_BUILD_EXECUTABLE "Configure UxAS using qualified dependency and LMCP packages" OFF)
if(UXAS_BUILD_EXECUTABLE)
    include(cmake/Uxas.cmake)
    return()
endif()
if(NOT EXISTS "${UXAS_LMCP_GENERATED_DIR}/avtas/lmcp/Factory.cpp" OR NOT EXISTS "${UXAS_LMCP_PROVENANCE}")
    message(FATAL_ERROR "Use build-lmcp-cpp.ps1 with a validated G1 generation batch")
endif()
include(cmake/LmcpSources.cmake)
add_library(uxas_lmcp STATIC ${_lmcp_sources})
add_library(Uxas::lmcp ALIAS uxas_lmcp)
set_target_properties(uxas_lmcp PROPERTIES EXPORT_NAME lmcp OUTPUT_NAME lmcp)
target_compile_features(uxas_lmcp PUBLIC cxx_std_14)
target_compile_definitions(uxas_lmcp PUBLIC WIN32)
target_compile_options(uxas_lmcp PRIVATE /W3 /utf-8 /MP8 /Bv)
target_include_directories(uxas_lmcp PUBLIC
    "$<BUILD_INTERFACE:${UXAS_LMCP_GENERATED_DIR}>" "$<INSTALL_INTERFACE:include>")
install(TARGETS uxas_lmcp EXPORT UxasLmcpTargets ARCHIVE DESTINATION lib COMPONENT Lmcp)
install(DIRECTORY "${UXAS_LMCP_GENERATED_DIR}/" DESTINATION include COMPONENT Lmcp FILES_MATCHING PATTERN "*.h" PATTERN test EXCLUDE)
install(EXPORT UxasLmcpTargets NAMESPACE Uxas:: DESTINATION share/UxasLmcp COMPONENT Lmcp)
install(FILES cmake/UxasLmcpConfig.cmake DESTINATION share/UxasLmcp COMPONENT Lmcp)
option(UXAS_LMCP_BUILD_PROBE "Build the file-only cross-language probe" ON)
if(UXAS_LMCP_BUILD_PROBE)
    add_executable(lmcp_probe tests/lmcp_cpp/probe.cpp)
    target_link_libraries(lmcp_probe PRIVATE Uxas::lmcp)
    target_compile_options(lmcp_probe PRIVATE /W4 /utf-8 /showIncludes)
    target_link_options(lmcp_probe PRIVATE /VERBOSE:LIB)
endif()
