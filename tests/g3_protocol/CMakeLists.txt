cmake_minimum_required(VERSION 3.31)
project(G3ProtocolObservation LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native MSVC x64 required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_LIBRARIES OFF)
find_package(UxasDependencies CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${UXAS_DEPENDENCIES_PREFIX}/share/UxasDependencies")
add_executable(pub_capture pub_capture.cpp)
target_compile_options(pub_capture PRIVATE /W4 /utf-8)
target_link_libraries(pub_capture PRIVATE UxasDeps::cppzmq)
