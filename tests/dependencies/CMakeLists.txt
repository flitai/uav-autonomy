cmake_minimum_required(VERSION 3.31)
project(UxasDependencyProbes LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The dependency probes require native MSVC x64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# CMake 3.31/Ninja writes a BOM-less UTF-8 response file, which link.exe reads
# using the system ACP. Keep Unicode library paths on the Unicode command line;
# object paths in the response file are relative ASCII names for this probe.
if(CMAKE_GENERATOR STREQUAL "Ninja")
    set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_LIBRARIES OFF)
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
find_package(UxasDependencies CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${UXAS_DEPENDENCIES_PREFIX}/share/UxasDependencies")
add_executable(deps_probe deps_probe.cpp)
target_compile_options(deps_probe PRIVATE /W4 /utf-8 /Bv /showIncludes)
target_link_options(deps_probe PRIVATE /VERBOSE:LIB)
target_link_libraries(deps_probe PRIVATE UxasDeps::cppzmq UxasDeps::czmq UxasDeps::pugixml UxasDeps::sqlitecpp UxasDeps::boost)
enable_testing()
foreach(component zmq pugi sqlite boost)
    add_test(NAME ${component} COMMAND deps_probe ${component})
    set_tests_properties(${component} PROPERTIES TIMEOUT 20 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
endforeach()
