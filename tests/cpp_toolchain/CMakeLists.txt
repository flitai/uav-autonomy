cmake_minimum_required(VERSION 3.31)
cmake_policy(SET CMP0091 NEW)
project(uav_toolchain_probe LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The probe requires native MSVC x64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(c_probe c_probe.c)
add_executable(cpp_probe cpp_probe.cpp)
foreach(target c_probe cpp_probe)
  target_compile_options(${target} PRIVATE /W4 /Bv /showIncludes)
  target_link_options(${target} PRIVATE /VERBOSE:LIB)
endforeach()
enable_testing()
add_test(NAME c_probe COMMAND c_probe)
add_test(NAME cpp_probe COMMAND cpp_probe)
