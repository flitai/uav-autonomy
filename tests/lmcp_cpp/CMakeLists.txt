cmake_minimum_required(VERSION 3.31)
project(LmcpConsumer LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native MSVC x64 required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_GENERATOR STREQUAL "Ninja")
    set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_LIBRARIES OFF)
endif()
find_package(UxasLmcp CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${UXAS_LMCP_PREFIX}/share/UxasLmcp")
add_executable(lmcp_probe probe.cpp)
target_link_libraries(lmcp_probe PRIVATE Uxas::lmcp)
target_compile_options(lmcp_probe PRIVATE /W4 /utf-8 /showIncludes)
target_link_options(lmcp_probe PRIVATE /VERBOSE:LIB)
