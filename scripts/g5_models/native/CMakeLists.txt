cmake_minimum_required(VERSION 3.31)
project(G5ModelReader LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "G5 model reader requires native MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /DNOMINMAX /utf-8")
set(BUILD_OSG_APPLICATIONS OFF CACHE BOOL "" FORCE)
set(BUILD_OSG_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_OSG_PLUGINS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(BUILD_OSG_PLUGIN_OSG ON CACHE BOOL "" FORCE)
set(BUILD_OSG_DEPRECATED_SERIALIZERS OFF CACHE BOOL "" FORCE)
set(OSG_USE_UTF8_FILENAME ON CACHE BOOL "" FORCE)
set(OSG_MSVC_VERSIONED_DLL OFF CACHE BOOL "" FORCE)
set(OSG_DETERMINE_WIN_VERSION OFF CACHE BOOL "" FORCE)
set(DYNAMIC_OPENSCENEGRAPH ON CACHE BOOL "" FORCE)
set(DYNAMIC_OPENTHREADS ON CACHE BOOL "" FORCE)
add_subdirectory("${OSG_SOURCE}" osg)
add_executable(g5-model-reader reader.cpp)
target_include_directories(g5-model-reader PRIVATE "${OSG_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/osg/include")
target_link_libraries(g5-model-reader PRIVATE osgDB osgUtil osg OpenThreads)
set_target_properties(g5-model-reader PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
